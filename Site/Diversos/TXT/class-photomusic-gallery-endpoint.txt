<?php
// includes/class-photomusic-gallery-endpoint.php

if (!defined('ABSPATH')) exit;

class PhotoMusic_Gallery_Endpoint {

    public static function init() {
        add_action('init', [__CLASS__, 'add_rewrite_rules']);
        add_action('template_redirect', [__CLASS__, 'handle_gallery_request']);
    }

    /**
     * Cria a rota:
     * /galeria/{slug_servico}/?token=XYZ
     */
    public static function add_rewrite_rules() {
        add_rewrite_rule(
            '^galeria/([^/]+)/?',
            'index.php?pm_galeria_servico=$matches[1]',
            'top'
        );

        add_rewrite_tag('%pm_galeria_servico%', '([^&]+)');
    }

    /**
     * Processa o endpoint
     */
    public static function handle_gallery_request() {
        $slug_servico = get_query_var('pm_galeria_servico');

        if (!$slug_servico) {
            return; // não é o endpoint
        }

        global $wpdb;

        $tbl_servicos = $wpdb->prefix . 'pm_event_services';
        $tbl_eventos  = $wpdb->prefix . 'pm_eventos';

        /* ============================================================
           1. Carregar serviço pelo slug
        ============================================================ */
        $servico = $wpdb->get_row($wpdb->prepare(
            "SELECT * FROM $tbl_servicos WHERE slug_servico = %s",
            sanitize_title($slug_servico)
        ));

        if (!$servico) {
            wp_die('Serviço não encontrado.');
        }

        $id_evento  = intval($servico->id_evento);
        $id_servico = intval($servico->id);

        /* ============================================================
           2. Carregar evento
        ============================================================ */
        $evento = $wpdb->get_row($wpdb->prepare(
            "SELECT * FROM $tbl_eventos WHERE id = %d",
            $id_evento
        ));

        if (!$evento) {
            wp_die('Evento não encontrado.');
        }

        if ($evento->status_evento === 'desativado') {
            wp_die('Este evento está encerrado.');
        }

        /* ============================================================
           3. Validar token
        ============================================================ */
        $token = isset($_GET['token']) ? sanitize_text_field($_GET['token']) : '';

        if (empty($token)) {
            wp_die('Token de acesso ausente.');
        }

        /* ============================================================
           4. Validar dispositivo
        ============================================================ */
        $device_hash = self::generate_device_hash();

        /* ============================================================
           5. Validar regras de acesso
        ============================================================ */
        $validacao = PhotoMusic_Access_Rules::validate_guest_access(
            $id_evento,
            $id_servico,
            $token,
            $device_hash
        );

        if (is_wp_error($validacao)) {
            wp_die($validacao->get_error_message());
        }

        /* ============================================================
           6. Carregar galeria protegida
        ============================================================ */
        $pasta = rtrim($servico->pasta_protegida, '/');

        if (!is_dir($pasta)) {
            wp_die('Nenhuma mídia encontrada para este serviço.');
        }

        $arquivos = glob($pasta . '/*');

        /* ============================================================
           7. Renderizar página da galeria
        ============================================================ */
        self::render_gallery_page($evento, $servico, $arquivos);

        exit;
    }

    /**
     * Gera um hash único baseado no dispositivo do usuário
     */
    private static function generate_device_hash() {
        $ua = $_SERVER['HTTP_USER_AGENT'] ?? '';
        $ip = $_SERVER['REMOTE_ADDR'] ?? '';

        return md5($ua . $ip);
    }

    /**
     * Renderiza a página HTML da galeria
     */
    private static function render_gallery_page($evento, $servico, $arquivos) {
        ?>
        <html>
        <head>
            <title><?php echo esc_html($evento->motivo_evento); ?> - <?php echo esc_html($servico->nome_servico); ?></title>
            <style>
                body { font-family: Arial; background: #f5f5f5; padding: 20px; }
                h1 { margin-bottom: 5px; }
                .grid { display: flex; flex-wrap: wrap; gap: 15px; }
                .item { width: 200px; background: #fff; padding: 10px; border-radius: 8px; text-align: center; }
                img, video { width: 100%; border-radius: 6px; }
            </style>
        </head>
        <body>

        <h1><?php echo esc_html($evento->motivo_evento); ?></h1>
        <h3><?php echo esc_html($servico->nome_servico); ?></h3>

        <div class="grid">
            <?php foreach ($arquivos as $arquivo): ?>
                <div class="item">
                    <?php if (preg_match('/\.(jpg|jpeg|png|gif)$/i', $arquivo)): ?>
                        <img src="<?php echo esc_url(self::secure_file_url($arquivo)); ?>">
                    <?php elseif (preg_match('/\.(mp4|mov|webm)$/i', $arquivo)): ?>
                        <video controls>
                            <source src="<?php echo esc_url(self::secure_file_url($arquivo)); ?>">
                        </video>
                    <?php endif; ?>
                </div>
            <?php endforeach; ?>
        </div>

        </body>
        </html>
        <?php
    }

    /**
     * Gera URL segura para entregar o arquivo via endpoint protegido
     */
    private static function secure_file_url($arquivo) {
        $encoded = base64_encode($arquivo);
        return home_url("/pm-file/?f=$encoded");
    }
}
