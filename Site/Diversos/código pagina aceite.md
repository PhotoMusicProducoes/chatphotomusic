(function () {
  const params   = new URLSearchParams(location.search);
  const tEvento  = params.get('t')   || '';
  const telefone = params.get('tel') || '';

  const elLoading   = document.getElementById('pm-loading');
  const elFormArea  = document.getElementById('pm-form-area');
  const elErro      = document.getElementById('pm-erro');
  const elErroMsg   = document.getElementById('pm-erro-msg');
  const elNomeEvt   = document.getElementById('pm-evento-nome');
  const elMsg       = document.getElementById('pm-msg');
  const elBtn       = document.getElementById('pm-btn');

  // Guarda dados retornados pelo /entrada para usar no submit
  let idEvento = 0;

  function mostrarErro(msg) {
    elLoading.style.display  = 'none';
    elFormArea.style.display = 'none';
    elErro.style.display     = 'block';
    elErroMsg.textContent    = msg;
  }

  // Passo 1: verifica se já aceitou
  if (!tEvento || !telefone) {
    mostrarErro('Link inválido. Use o link enviado pelo WhatsApp.');
    return;
  }

  const apiBase = (window.pm_settings && pm_settings.api_url)
    ? pm_settings.api_url
    : location.origin + '/wp-json/photomusic/v1';

  fetch(apiBase + '/entrada?t=' + encodeURIComponent(tEvento) + '&tel=' + encodeURIComponent(telefone))
    .then(r => r.json())
    .then(data => {

      if (data.acao === 'galeria') {
        // Já aceitou — redireciona direto
        location.replace(data.redirect);
        return;
      }

      if (data.acao === 'aceite') {
        // Precisa aceitar — mostra formulário
        idEvento = data.id_evento;
        elNomeEvt.textContent    = data.nome || 'Evento';
        elLoading.style.display  = 'none';
        elFormArea.style.display = 'block';
        return;
      }

      // acao === 'erro'
      mostrarErro(data.mensagem || 'Link inválido ou expirado.');
    })
    .catch(() => mostrarErro('Não foi possível verificar o link. Tente novamente.'));

  // Passo 2: submit do formulário
  document.getElementById('pm-aceite-form').addEventListener('submit', function (e) {
    e.preventDefault();

    const nome  = document.getElementById('pm-nome').value.trim();
    const email = document.getElementById('pm-email').value.trim();

    if (!nome) {
      elMsg.style.display = 'block';
      elMsg.style.color   = '#c00';
      elMsg.textContent   = 'Por favor, informe seu nome.';
      return;
    }

    elBtn.disabled     = true;
    elBtn.textContent  = 'Aguarde...';
    elMsg.style.display = 'none';

    const idioma = navigator.language ? navigator.language.substring(0, 2) : 'pt';

    fetch(apiBase + '/aceite', {
      method:  'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        nome:     nome,
        email:    email,
        telefone: telefone,
        idioma:   idioma,
        idEvento: idEvento
      })
    })
    .then(r => r.json())
    .then(data => {
      if (data.sucesso && data.redirect) {
        location.replace(data.redirect);
      } else {
        elMsg.style.display = 'block';
        elMsg.style.color   = '#c00';
        elMsg.textContent   = data.mensagem || 'Ocorreu um erro. Tente novamente.';
        elBtn.disabled    = false;
        elBtn.textContent = 'Aceitar e Acessar as Fotos';
      }
    })
    .catch(() => {
      elMsg.style.display = 'block';
      elMsg.style.color   = '#c00';
      elMsg.textContent   = 'Erro de conexão. Tente novamente.';
      elBtn.disabled    = false;
      elBtn.textContent = 'Aceitar e Acessar as Fotos';
    });
  });
})();
</script>
